{{- /* Language switcher: links to the current page's translation per language, falling back to that language's home (derived from the home page's translations, so no dependency on the default-content-language config). */ -}} {{- if gt (len $.Site.Languages) 1 -}} {{- $homes := dict $.Site.Language.Lang $.Site.Home.RelPermalink -}} {{- range $.Site.Home.Translations -}} {{- $homes = merge $homes (dict .Language.Lang .RelPermalink) -}} {{- end -}}
{{- range $.Site.Languages -}} {{- $lang := .Lang -}} {{- $name := .Label | default (upper $lang) -}} {{- if eq $lang $.Site.Language.Lang -}} {{ $name }} {{- else -}} {{- $target := index $homes $lang | default (printf "/%s/" $lang | absURL) -}} {{- with $.Translations -}} {{- range $.Translations -}} {{- if eq .Language.Lang $lang -}} {{- $target = .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{ $name }} {{- end -}} {{- end -}}
{{- end -}}
