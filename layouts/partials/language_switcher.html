{{- /* Language switcher: links to the current page's translation per language, falling back to that language's home. Enumerates languages via hugo.Sites, the modern replacement for the deprecated .Site.Languages. */ -}} {{- if gt (len hugo.Sites) 1 -}} {{- /* Build a lang -> home permalink map across all sites. */ -}} {{- $homes := dict -}} {{- range hugo.Sites -}} {{- $homes = merge $homes (dict .Language.Lang .Home.RelPermalink) -}} {{- end -}}
{{- range hugo.Sites -}} {{- $lang := .Language.Lang -}} {{- $name := .Language.Label | default (upper $lang) -}} {{- if eq $lang $.Site.Language.Lang -}} {{ $name }} {{- else -}} {{- $target := index $homes $lang -}} {{- with $.Translations -}} {{- range $.Translations -}} {{- if eq .Language.Lang $lang -}} {{- $target = .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{ $name }} {{- end -}} {{- end -}}
{{- end -}}
